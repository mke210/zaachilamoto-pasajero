// ========== REEMPLAZAR FUNCIÓN escucharMiServicio ========== function escucharMiServicio(serviceId) { database.ref(`servicios/${serviceId}`).on('value', (snap) => { const servicio = snap.val(); if (!servicio) return; console.log('📋 Estado:', servicio.estado); if (servicio.estado === 'tarifa_propuesta') { // Chofer aceptó y propuso tarifa - mostrar al cliente document.getElementById('activeDriverName').textContent = servicio.choferNombre || 'Chofer'; document.getElementById('activeDriverPlate').textContent = 'Placa: ' + (servicio.choferPlaca || '---'); document.getElementById('activeDriverRating').textContent = '⭐ ' + (servicio.choferRating || 'Nuevo'); document.getElementById('etaDisplay').textContent = 'Por confirmar'; document.getElementById('fareDisplay').textContent = '$' + (servicio.tarifa_propuesta || servicio.tarifa_estimada || '40'); document.getElementById('distanceDisplay').textContent = '~1 km'; // Mostrar modal de aceptación de tarifa mostrarConfirmacionTarifa(serviceId, servicio); } else if (servicio.estado === 'aceptado') { document.getElementById('stepCircle1').classList.add('active'); document.getElementById('etaDisplay').textContent = '3-5 min'; showToast('✅ Tarifa aceptada - Chofer en camino'); } else if (servicio.estado === 'en_camino') { document.getElementById('stepCircle1').classList.add('completed'); document.getElementById('stepCircle2').classList.add('active'); document.getElementById('etaDisplay').textContent = '1-2 min'; } else if (servicio.estado === 'en_servicio') { document.getElementById('stepCircle2').classList.add('completed'); document.getElementById('stepCircle3').classList.add('active'); document.getElementById('etaDisplay').textContent = 'En viaje...'; } else if (servicio.estado === 'completado') { document.getElementById('stepCircle3').classList.add('completed'); showToast('✅ ¡Viaje completado!'); setTimeout(resetearApp, 5000); } else if (servicio.estado === 'cancelado' || servicio.estado === 'rechazado') { showToast('❌ Servicio no disponible. Busca otro mototaxi.'); resetearApp(); } }); } // ========== AGREGAR FUNCIÓN DE CONFIRMACIÓN DE TARIFA ========== function mostrarConfirmacionTarifa(serviceId, servicio) { const tarifa = servicio.tarifa_propuesta || servicio.tarifa_estimada || 40; const choferNombre = servicio.choferNombre || 'Chofer'; if (confirm('💰 ' + choferNombre + ' aceptó tu servicio.\n\nTarifa propuesta: $' + tarifa + ' MXN\n\n¿Aceptas esta tarifa?')) { // Cliente acepta la tarifa database.ref(`servicios/${serviceId}`).update({ estado: 'aceptado', cliente_acepta_tarifa: true, timestamp_aceptacion: firebase.database.ServerValue.TIMESTAMP }).then(() => { showToast('✅ Tarifa aceptada - Chofer viene en camino'); }); } else { // Cliente rechaza la tarifa database.ref(`servicios/${serviceId}`).update({ estado: 'cancelado', motivo: 'Cliente rechazó tarifa', timestamp_cancelacion: firebase.database.ServerValue.TIMESTAMP }).then(() => { showToast('❌ Servicio cancelado'); resetearApp(); }); } }
