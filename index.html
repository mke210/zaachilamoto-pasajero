// ========== CLIENTE: REEMPLAZAR escucharMiServicio ========== function escucharMiServicio(serviceId) { database.ref(`servicios/${serviceId}`).on('value', (snap) => { const servicio = snap.val(); if (!servicio) return; console.log('📋 Estado recibido:', servicio.estado, 'Tarifa:', servicio.tarifa_propuesta); if (servicio.estado === 'tarifa_propuesta') { // Mostrar datos del chofer document.getElementById('activeDriverName').textContent = servicio.choferNombre || 'Chofer'; document.getElementById('activeDriverPlate').textContent = 'Placa: ' + (servicio.choferPlaca || '---'); document.getElementById('activeDriverRating').textContent = '⭐ ' + (servicio.choferRating || 'Nuevo'); const tarifa = servicio.tarifa_propuesta || servicio.tarifa_estimada || 40; document.getElementById('fareDisplay').textContent = '$' + tarifa; document.getElementById('etaDisplay').textContent = 'Por confirmar'; document.getElementById('distanceDisplay').textContent = '~1 km'; // Mostrar confirmación de tarifa mostrarConfirmacionTarifa(serviceId, servicio, tarifa); } else if (servicio.estado === 'aceptado') { document.getElementById('stepCircle1').classList.add('active'); document.getElementById('etaDisplay').textContent = '3-5 min'; showToast('✅ Chofer en camino'); } else if (servicio.estado === 'en_camino') { document.getElementById('stepCircle1').classList.add('completed'); document.getElementById('stepCircle2').classList.add('active'); document.getElementById('etaDisplay').textContent = '1-2 min'; } else if (servicio.estado === 'en_servicio') { document.getElementById('stepCircle2').classList.add('completed'); document.getElementById('stepCircle3').classList.add('active'); document.getElementById('etaDisplay').textContent = 'En viaje...'; } else if (servicio.estado === 'completado') { document.getElementById('stepCircle3').classList.add('completed'); showToast('✅ ¡Viaje completado!'); setTimeout(resetearApp, 5000); } else if (servicio.estado === 'cancelado' || servicio.estado === 'rechazado') { showToast('❌ Servicio no disponible'); resetearApp(); } }); } // ========== CLIENTE: AGREGAR CONFIRMACIÓN DE TARIFA CON MODAL ========== function mostrarConfirmacionTarifa(serviceId, servicio, tarifa) { const choferNombre = servicio.choferNombre || 'Chofer'; const choferPlaca = servicio.choferPlaca || '---'; const choferRating = servicio.choferRating || 'Nuevo'; // Crear modal de confirmación const modal = document.createElement('div'); modal.style.cssText = ` position: fixed; top: 0; left: 0; width: 100%; height: 100%; background: rgba(0,0,0,0.85); z-index: 9999; display: flex; align-items: center; justify-content: center; `; modal.innerHTML = `
🏍️

${choferNombre}

${choferPlaca} · ⭐ ${choferRating}

TARIFA PROPUESTA
$${tarifa} MXN
Pago en efectivo

⚠️ Confirma para que el chofer vaya a recogerte

`; document.body.appendChild(modal); // Eventos de los botones modal.querySelector('#btnAcceptFare').onclick = () => { modal.remove(); database.ref(`servicios/${serviceId}`).update({ estado: 'aceptado', cliente_acepta_tarifa: true, timestamp_aceptacion: firebase.database.ServerValue.TIMESTAMP }).then(() => { showToast('✅ Tarifa aceptada'); }); }; modal.querySelector('#btnRejectFare').onclick = () => { modal.remove(); database.ref(`servicios/${serviceId}`).update({ estado: 'cancelado', motivo: 'Cliente rechazó tarifa', timestamp_cancelacion: firebase.database.ServerValue.TIMESTAMP }).then(() => { showToast('❌ Servicio cancelado'); resetearApp(); }); }; // También cerrar al hacer clic fuera modal.onclick = (e) => { if (e.target === modal) { // No cerrar, obligar a elegir } }; }
